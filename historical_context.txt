export function createSearchIndex(tools: Tool[]): Fuse<Tool> {
  fuseInstance = new Fuse(tools, {
    keys: [

function expandQuery(query: string): string[] {
  const lower = query.toLowerCase().trim();
  const expanded = [lower];

  // Check task synonyms
  for (const [task, synonyms] of Object.entries(TASK_SYNONYMS)) {
    if (lower.includes(task) || synonyms.some(s => lower.includes(s))) {
      expanded.push(task, ...synonyms);
    }
  }

  // Check category keywords
  for (const [category, keywords] of Object.entries(CATEGORY_KEYWORDS)) {
    if (keywords.some(k => lower.includes(k) || k.includes(lower))) {
      expanded.push(category);
    }
  }

  return [...new Set(expanded)];
}

export function searchTools(
  tools: Tool[],
  fuse: Fuse<Tool>,
  query: string,
  options: {
    access?: string;
    category?: string;
    sort?: SortOption;
    studentMode?: boolean;
  } = {}
): Tool[] {
  let results: Tool[];

  if (!query.trim()) {
    results = [...tools];
  } else {
    const expandedTerms = expandQuery(query);

    // Search with each expanded term and merge results
    const scoreMap = new Map<number, number>();

    for (const term of expandedTerms) {
      const fuseResults = fuse.search(term);
      for (const r of fuseResults) {
        const existing = scoreMap.get(r.item.id) ?? 1;
        const newScore = Math.min(existing, r.score ?? 1);
        scoreMap.set(r.item.id, newScore);
      }
    }

    if (scoreMap.size === 0) {
      results = [];
    } else {
      results = tools
        .filter(t => scoreMap.has(t.id))
        .sort((a, b) => (scoreMap.get(a.id) ?? 1) - (scoreMap.get(b.id) ?? 1));
    }
  }

  // Apply student mode filter
  if (options.studentMode) {
    results = results.filter(t => t.access !== 'Paid');
  }

  // Apply access filter
  if (options.access && options.access !== 'all') {
    results = results.filter(t => t.access === options.access);
  }

  // Apply category filter
  if (options.category && options.category !== 'all') {
    results = results.filter(t => t.category === options.category);
  }

  // Apply sort
  if (options.sort === 'alphabetical') {
    results = [...results].sort((a, b) => a.name.localeCompare(b.name));
  } else if (options.sort === 'recently-added') {
    results = [...results].sort((a, b) => b.id - a.id);
  }

  return results;
}

  bestOverall: Tool[];

  bestForStudents: Tool[];
  bestPremium: Tool[];
  mostPopular: Tool[];
  fastestToLearn: Tool[];

  // "Most popular" — use a fixed popularity list of well-known tools

  const mostPopular = tools.filter(t => popularNames.includes(t.name)).slice(0, topN);

  };

Business Rules and Logic\\n\\n### 4.1 Data Import and Parsing\\n\\n- Import entire PDF containing 500 AI tools\\n- Parse each tool entry and extract: Tool Name, Official Website, Category, Why Use It, Access Type (Free, Freemium, Paid, Open Source), Free Plan Limits\\n- Automatically categorize tools into 25 categories\\n- Index all tool data for search functionality\\n- Never hardcode only a few tools; entire PDF must be parsed and displayed accurately\\n\\n### 4.2 Search Logic\\n\\n- User searches for tasks, not tool names\\n- Example: User types \\\\"Create PowerPoint Presentation\\\\", \\\\"Make PPT\\\\", \\\\"Presentation\\\\", or \\\\"Slides\\\\"\\n- System intelligently understands intent and recommends tools: Gamma, Canva, Beautiful.ai, Tome, SlidesAI, Plus AI, and any other relevant tools from PDF\\n- Support natural language queries, synonyms, fuzzy matching, typo tolerance\\n- Provide autocomplete suggestions based on common tasks\\n- Display recent searches and trending searches\\n- Apply instant filtering without page reload\\n\\n### 4.3 Ranking and Recommendation\\n\\n- Rank search results into sections: Best Overall, Best Free, Best for Students, Best Premium, Most Popular, Fastest to Learn\\n- Ranking criteria:\\n  - Best Overall: highest relevance to user query\\n  - Best Free: tools with Access Type = Free\\n  - Best for Students: tools with Access Type = Free or Freemium\\n  - Best Premium: tools with Access Type = Paid\\n  - Most Popular: tools with highest usage or bookmark count\\n  - Fastest to Learn: tools with simplest onboarding or interface\\n\\n### 4.4 Student Mode\\n\\n- Provide toggle to enable Student Mode\\n- When enabled:\\n  - Prioritize Free and Freemium tools in search results\\n  - Hide Paid tools by default\\n  - Display \\\\"Best Free Choice\\\\" badge on recommended tools\\n- When disabled:\\n  - Display all tools regardless of Access Type\\n\\n### 4.5 Comparison Logic\\n\\n- Allow user to select up to 3 tools for comparison\\n- Display comparison table with: Tool Name, Category, Access Type, Free Plan Limits, Why Use It, Official Website\\n- Highlight differences visually (e.g., different Access Types, different Free Plan Limits)\\n- Prevent user from adding more than 3 tools to comparison\\n\\n### 4.6 Bookmark and History\\n\\n- Store bookmarked tools in browser local storage\\n- Persist bookmarks after page refresh\\n- Store recently viewed tools in browser local storage\\n- Display recently viewed tools in Bookmarks Page\\n- Limit recently viewed list to 20 most recent tools\\n\\n### 4.7 Filtering and Sorting\\n\\n- Filters: Free Only, Freemium, Paid, Open Source, Category (25 categories), Sort (Alphabetical, Popularity, Recently Added)\\n- Apply filters instantly without page reload\\n- Allow multiple filters to be active simultaneously\\n- Sorting options:\\n  - Alphabetical: sort by Tool Name A-Z\\n  - Popularity: sort by bookmark count or usage\\n  - Recently Added: sort by date added to database\\n\\n---\\n\\n## 5. import { useState, useCallback } from 'react';
import type { Tool } from '@/types/tool';

const BOOKMARKS_KEY = 'toolverse_bookmarks';
const HISTORY_KEY = 'toolverse_history';
const MAX_HISTORY = 20;

function loadFromStorage<T>(key: string, fallback: T): T {
  try {
    const raw = localStorage.getItem(key);
    return raw ? JSON.parse(raw) : fallback;
  } catch {
    return fallback;
  }
}

function saveToStorage<T>(key: string, value: T): void {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch {
    // Silently fail if storage is unavailable
  }
}

export function useBookmarks() {
  const [bookmarks, setBookmarks] = useState<Tool[]>(() =>
    loadFromStorage<Tool[]>(BOOKMARKS_KEY, [])
  );

  const isBookmarked = useCallback(
    (id: number) => bookmarks.some(t => t.id === id),
    [bookmarks]
  );

  const toggleBookmark = useCallback((tool: Tool) => {
    setBookmarks(prev => {
      const exists = prev.some(t => t.id === tool.id);
      const next = exists ? prev.filter(t => t.id !== tool.id) : [tool, ...prev];
      saveToStorage(BOOKMARKS_KEY, next);
      return next;
    });
  }, []);

  const clearBookmarks = useCallback(() => {
    setBookmarks([]);
    saveToStorage(BOOKMARKS_KEY, []);
  }, []);

  return { bookmarks, isBookmarked, toggleBookmark, clearBookmarks };
}

export function useRecentlyViewed() {
  const [recentlyViewed, setRecentlyViewed] = useState<Tool[]>(() =>
    loadFromStorage<Tool[]>(HISTORY_KEY, [])
  );

  const addToHistory = useCallback((tool: Tool) => {
    setRecentlyViewed(prev => {
      const filtered = prev.filter(t => t.id !== tool.id);
      const next = [tool, ...filtered].slice(0, MAX_HISTORY);
      saveToStorage(HISTORY_KEY, next);
      return next;
    });
  }, []);

  const clearHistory = useCallback(() => {
    setRecentlyViewed([]);
    saveToStorage(HISTORY_KEY, []);
  }, []);

  return { recentlyViewed, addToHistory, clearHistory };
}

export function useCompare() {
  const [compareList, setCompareList] = useState<Tool[]>([]);

  const isInCompare = useCallback(
    (id: number) => compareList.some(t => t.id === id),
    [compareList]
  );

  const toggleCompare = useCallback((tool: Tool) => {
    setCompareList(prev => {
      if (prev.some(t => t.id === tool.id)) {
        return prev.filter(t => t.id !== tool.id);
      }
      if (prev.length >= 3) return prev;
      return [...prev, tool];
    });
  }, []);

  const removeFromCompare = useCallback((id: number) => {
    setCompareList(prev => prev.filter(t => t.id !== id));
  }, []);

  const clearCompare = useCallback(() => setCompareList([]), []);

  return { compareList, isInCompare, toggleCompare, removeFromCompare, clearCompare };
}

const SEARCHES_KEY = 'toolverse_recent_searches';
const MAX_SEARCHES = 8;

export function useRecentSearches() {
  const [recentSearches, setRecentSearches] = useState<string[]>(() =>
    loadFromStorage<string[]>(SEARCHES_KEY, [])
  );

  const addSearch = useCallback((query: string) => {
    const trimmed = query.trim();
    if (!trimmed) return;
    setRecentSearches(prev => {
      const filtered = prev.filter(s => s.toLowerCase() !== trimmed.toLowerCase());
      const next = [trimmed, ...filtered].slice(0, MAX_SEARCHES);
      saveToStorage(SEARCHES_KEY, next);
      return next;
    });
  }, []);

  const removeSearch = useCallback((query: string) => {
    setRecentSearches(prev => {
      const next = prev.filter(s => s !== query);
      saveToStorage(SEARCHES_KEY, next);
      return next;
    });
  }, []);

  const clearSearches = useCallback(() => {
    setRecentSearches([]);
    saveToStorage(SEARCHES_KEY, []);
  }, []);

  return { recentSearches, addSearch, removeSearch, clearSearches };
}

  'Automate workflows…',
  'Analyze data with AI…',

interface Props {

  initialValue?: string;
  onSearch?: (query: string) => void;

export default function SearchBar({ size = 'hero', autoFocus = false, initialValue = '', onSearch }: Props) {
  const navigate = useNavigate();
  const { recentSearches, addSearch, removeSearch } = useApp();
  const [value, setValue] = useState(initialValue);
  const [focused, setFocused] = useState(false);
  const [placeholderIdx, setPlaceholderIdx] = useState(0);
  const [suggestions, setSuggestions] = useState<string[]>([]);
  const [selectedIdx, setSelectedIdx] = useState(-1);
  const inputRef = useRef<HTMLInputElement>(null);
  const dropdownRef = useRef<HTMLDivElement>(null);

  // Rotate placeholder
  useEffect(() => {
    const id = setInterval(() => {
      setPlaceholderIdx(i => (i + 1) % PLACEHOLDERS.length);
    }, 2800);
    return () => clearInterval(id);
  }, []);

  // Update suggestions
  useEffect(() => {
    setSuggestions(value.trim() ? getAutocompleteSuggestions(value) : []);
    setSelectedIdx(-1);
  }, [value]);

  useEffect(() => {
    setValue(initialValue);
  }, [initialValue]);

  const handleSubmit = useCallback((q: string) => {
    const trimmed = q.trim();
    if (!trimmed) return;
    addSearch(trimmed);
    if (onSearch) {
      onSearch(trimmed);
    } else {
      navigate(`/search?q=${encodeURIComponent(trimmed)}`);
    }
    setFocused(false);
    inputRef.current?.blur();
  }, [addSearch, navigate, onSearch]);

  const handleKeyDown = (e: React.KeyboardEvent) => {
    const items = value.trim() ? suggestions : [...recentSearches.slice(0, 5), ...TRENDING.slice(0, 3)];
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setSelectedIdx(i => Math.min(i + 1, items.length - 1));
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setSelectedIdx(i => Math.max(i - 1, -1));
    } else if (e.key === 'Enter') {
      e.preventDefault();
      if (selectedIdx >= 0 && items[selectedIdx]) {
        handleSubmit(items[selectedIdx]);
      } else {
        handleSubmit(value);
      }
    } else if (e.key === 'Escape') {
      setFocused(false);
    }
  };

  const showDropdown = focused && (value.trim() ? suggestions.length > 0 : (recentSearches.length > 0 || TRENDING.length > 0));

  const isHero = size === 'hero';

  return (
    <div className="relative w-full">
      <div
        className={cn(
          'relative flex items-center rounded-2xl border transition-all duration-200',
          isHero ? 'h-14 md:h-16' : 'h-12',
          focused
            ? 'border-primary/40 bg-card shadow-hover'
            : 'border-white/[0.08] bg-card hover:border-white/[0.12]',
        )}
      >

        <input
          ref={inputRef}
          type="text"
          value={value}
          autoFocus={autoFocus}
          onChange={e => setValue(e.target.value)}
          onFocus={() => setFocused(true)}
          onBlur={() => setTimeout(() => setFocused(false), 150)}
          onKeyDown={handleKeyDown}
          placeholder={PLACEHOLDERS[placeholderIdx]}
          className={cn(
            'w-full bg-transparent text-foreground placeholder:text-muted-foreground/50 outline-none',
            isHero ? 'pl-12 pr-32 text-base' : 'pl-10 pr-24 text-sm',
          )}
        />
        {value && (
          <button
            onClick={() => { setValue(''); inputRef.current?.focus(); }}
            className="absolute right-24 text-muted-foreground hover:text-foreground transition-colors"
          >
            <X className="w-4 h-4" />
          </button>
        )}
        <button
          onClick={() => handleSubmit(value)}
          className={cn(
            'absolute right-2 flex items-center gap-1.5 bg-primary hover:bg-primary/90 text-primary-foreground font-medium rounded-xl transition-all active:scale-95',
            isHero ? 'px-4 py-2 text-sm h-10' : 'px-3 py-1.5 text-xs h-8'
          )}
        >
          Search <ArrowRight className={cn(isHero ? 'w-4 h-4' : 'w-3.5 h-3.5')} />
        </button>
      </div>

      {/* Dropdown */}
      {showDropdown && (
        <div
          ref={dropdownRef}
          className="absolute top-full left-0 right-0 mt-2 rounded-2xl bg-[#1B1B1B] border border-white/[0.08] overflow-hidden z-50"
          style={{ boxShadow: '0 16px 48px rgba(0,0,0,0.6)' }}
        >
          {value.trim() ? (
            <div className="p-2">
              {suggestions.map((s, i) => (
                <button
                  key={s}
                  onClick={() => { setValue(s); handleSubmit(s); }}
                  className={cn(
                    'flex items-center gap-3 w-full px-3 py-2.5 rounded-xl text-sm text-left transition-colors',
                    selectedIdx === i ? 'bg-white/[0.06] text-foreground' : 'text-muted-foreground hover:bg-white/[0.04] hover:text-foreground'
                  )}
                >
                  <Search className="w-3.5 h-3.5 shrink-0" />
                  {s}
                </button>
              ))}
            </div>
          ) : (
            <div className="p-2">
              {recentSearches.length > 0 && (
                <>
                  <div className="px-3 pt-1 pb-1 flex items-center justify-between">
                    <span className="text-[11px] font-semibold uppercase tracking-widest text-muted-foreground">Recent</span>

                  {recentSearches.slice(0, 4).map((s, i) => (
                    <button
                      key={s}
                      onClick={() => { setValue(s); handleSubmit(s); }}
                      className={cn(
                        'flex items-center gap-3 w-full px-3 py-2 rounded-xl text-sm text-left transition-colors',
                        selectedIdx === i ? 'bg-white/[0.06]' : 'text-muted-foreground hover:bg-white/[0.04] hover:text-foreground'
                      )}
                    >
                      <Clock className="w-3.5 h-3.5 shrink-0 text-muted-foreground/60" />
                      <span className="flex-1">{s}</span>
                      <X
                        className="w-3.5 h-3.5 opacity-0 group-hover:opacity-100 hover:opacity-100"
                        onClick={e => { e.stopPropagation(); removeSearch(s); }}

              <div className="px-3 pt-2 pb-1">

              ))}
            </div>
          )}
        </div>
      )}
    </div>
  );
}

    63       { name: 'name', weight: 3 },
    64       { name: 'why', weight: 2.5 },
    65       { name: 'category', weight: 2 },
    66       { name: 'access', weight: 0.5 },
    67       { name: 'freePlan', weight: 0.3 },
    68     ],
    69     threshold: 0.4,
    70     includeScore: true,
    71     useExtendedSearch: false,
    72     ignoreLocation: true,
    73     minMatchCharLength: 2,
    74   });
    75   return fuseInstance;
    76 }

    78 function expandQuery(query: string): string[] {
    79   const lower = query.toLowerCase().trim();
    80   const expanded = [lower];
    81 
    82   // Check task synonyms
    83   for (const [task, synonyms] of Object.entries(TASK_SYNONYMS)) {
    84     if (lower.includes(task) || synonyms.some(s => lower.includes(s))) {
    85       expanded.push(task, ...synonyms);
    86     }
    87   }
    88 
    89   // Check category keywords
    90   for (const [category, keywords] of Object.entries(CATEGORY_KEYWORDS)) {
    91     if (keywords.some(k => lower.includes(k) || k.includes(lower))) {
    92       expanded.push(category);
    93     }
    94   }
    95 
    96   return [...new Set(expanded)];
    97 }

    99 export function searchTools(
   100   tools: Tool[],
   101   fuse: Fuse<Tool>,
   102   query: string,
   103   options: {
   104     access?: string;
   105     category?: string;
   106     sort?: SortOption;
   107     studentMode?: boolean;
   108   } = {}
   109 ): Tool[] {
   110   let results: Tool[];
   111 
   112   if (!query.trim()) {
   113     results = [...tools];
   114   } else {
   115     const expandedTerms = expandQuery(query);
   116 
   117     // Search with each expanded term and merge results
   118     const scoreMap = new Map<number, number>();
   119 
   120     for (const term of expandedTerms) {
   121       const fuseResults = fuse.search(term);
   122       for (const r of fuseResults) {
   123         const existing = scoreMap.get(r.item.id) ?? 1;
   124         const newScore = Math.min(existing, r.score ?? 1);
   125         scoreMap.set(r.item.id, newScore);
   126       }
   127     }
   128 
   129     if (scoreMap.size === 0) {
   130       results = [];
   131     } else {
   132       results = tools
   133         .filter(t => scoreMap.has(t.id))
   134         .sort((a, b) => (scoreMap.get(a.id) ?? 1) - (scoreMap.get(b.id) ?? 1));
   135     }
   136   }
   137 
   138   // Apply student mode filter
   139   if (options.studentMode) {
   140     results = results.filter(t => t.access !== 'Paid');
   141   }
   142 
   143   // Apply access filter
   144   if (options.access && options.access !== 'all') {
   145     results = results.filter(t => t.access === options.access);
   146   }
   147 
   148   // Apply category filter
   149   if (options.category && options.category !== 'all') {
   150     results = results.filter(t => t.category === options.category);
   151   }
   152 
   153   // Apply sort
   154   if (options.sort === 'alphabetical') {
   155     results = [...results].sort((a, b) => a.name.localeCompare(b.name));
   156   } else if (options.sort === 'recently-added') {
   157     results = [...results].sort((a, b) => b.id - a.id);
   158   }
   159 
   160   return results;
   161 }

   163 export function rankResults(tools: Tool[], query: string): {
   164   bestOverall: Tool[];
   165   bestFree: Tool[];
   166   bestForStudents: Tool[];
   167   bestPremium: Tool[];
   168   mostPopular: Tool[];
   169   fastestToLearn: Tool[];
   170 } {
   171   const topN = Math.min(8, tools.length);
   172 
   173   const bestFree = tools.filter(t => t.access === 'Free' || t.access === 'Open Source').slice(0, topN);
   174   const bestForStudents = tools.filter(t => t.access !== 'Paid').slice(0, topN);
   175   const bestPremium = tools.filter(t => t.access === 'Paid').slice(0, topN);
   176 
   177   // "Most popular" — use a fixed popularity list of well-known tools
   178   const popularNames = ['ChatGPT', 'Claude', 'Canva', 'GitHub Copilot', 'Midjourney', 'Runway', 'Gamma', 'Notion', 'Figma', 'Grammarly', 'Perplexity', 'Google Gemini', 'CapCut', 'Adobe Firefly', 'ElevenLabs'];
   179   const mostPopular = tools.filter(t => popularNames.includes(t.name)).slice(0, topN);
   180 
   181   // "Fastest to learn" — freemium tools with simple why descriptions (short)
   182   const fastestToLearn = tools
   183     .filter(t => t.access !== 'Paid')
   184     .sort((a, b) => a.why.length - b.why.length)
   185     .slice(0, topN);
   186 
   187   return {
   188     bestOverall: tools.slice(0, topN),
   189     bestFree,
   190     bestForStudents,
   191     bestPremium,
   192     mostPopular: mostPopular.length ? mostPopular : tools.slice(0, topN),
   193     fastestToLearn,
   194   };
   195 }
